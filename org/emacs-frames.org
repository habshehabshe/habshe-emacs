#+title: habshe-emacs — frames
#+author: A. Shehab (habshehabshe)

Everything frame-level by mechanism: theme (faces), fonts (face
attributes applied with frame nil, which covers all future frames
including daemon ones), frame title, and tooltip/dialog
suppression.

Fonts: Adobe Courier for fixed-pitch faces, Adobe Helvetica for
=variable-pitch= (both bitmap fonts with identical scaling on this
machine). Default size 9pt, a real Courier bitmap size. Zoom steps
one point and does not snap to bitmap sizes; intermediate sizes
are synthesized scaling, accepted.

Fonts are set with direct face attributes (=:family=, =:height=),
never =font-spec=: =font-spec= has no =:height= property (its size
keyword is =:size=, an integer meaning pixels or a float meaning
points), and unknown keys are silently ignored — a spec with
=:height= requests the family at fontconfig's default size, and
=:font= then copies that opened size into the face. This bug is
why the config once started at 12pt until zoom-reset.

Only =default=, =fixed-pitch=, and =variable-pitch= are set
directly. =italic=, =bold=, and =bold-italic= inherit family and
height from =default= and carry their own weight/slant specs.

One font size everywhere: after the theme loads, every named
face's height is forced to the relative 1.0 — no face may declare
its own size. All sizing flows from the default face, so zooming
is one knob and it resizes the whole instance at once, like a
terminal. One known exception, investigated and left alone: the
startup screen's copyright line carries an anonymous inline face
spec =(variable-pitch (:height 0.8))= baked into the buffer text
by =fancy-startup-tail= — not a named face, so normalization
cannot reach it, and it is upstream's deliberate design.

Frame *creation* settings (bar modes, resize pair) deliberately
live in early-init, not here — they must be true before the first
frame is mapped or the bars flash. Do not migrate them down.

#+begin_src emacs-lisp :tangle "../lisp/my-emacs-frames.el"
  ;;; my-emacs-frames.el --- GENERATED from org/emacs-frames.org — do not edit -*- lexical-binding: t; -*-
  ;;; Commentary: Frame-level appearance: theme, fonts, title,
  ;;; tooltips and dialogs. One font size everywhere.

  (defgroup my-emacs-frames nil
    "Frame-level appearance for my Emacs configuration.")

  ;; --- Theme ---------------------------------------------------------------
  (require-theme 'modus-themes)
  (modus-themes-load-theme 'modus-operandi)

  ;; --- Fonts ---------------------------------------------------------------
  ;; Adobe Courier for fixed-pitch faces, Adobe Helvetica for
  ;; variable-pitch. Both bitmap; Courier's real sizes on this machine
  ;; are 6, 8, 9, 11, 13, 14, 15, 18, 19, 26 — the default (9pt) is one
  ;; of them.
  ;;
  ;; Set via direct face attributes, NOT font-spec: font-spec has no
  ;; :height property (its size keyword is :size — integer pixels or
  ;; float points), and unknown keys are silently stored as inert
  ;; extras. A spec carrying :height requests the family at
  ;; fontconfig's default size, and setting :font then copies that
  ;; opened size into the face's :height — the startup-at-12pt bug.
  ;; Face :height is an integer in 1/10 pt, matching our constants.
  ;;
  ;; Only these three faces are set directly; italic, bold, and
  ;; bold-italic inherit family and height from default.
  (defconst my-font-family "Adobe Courier"
    "Font family for fixed-pitch faces.")

  (defconst my-variable-pitch-font-family "Adobe Helvetica"
    "Font family for the variable-pitch face.")

  (defconst my-font-height-default 90
    "Default global font height in 1/10 pt (9pt, a real Courier bitmap size).")

  (defvar my-font-height my-font-height-default
    "Global font height in 1/10 pt. The only font-size knob.")

  (defconst my-font-height-step 10
    "Font height change per zoom step, in 1/10 pt (one point).")

  (set-face-attribute 'default nil
                      :family my-font-family
                      :height my-font-height)
  (set-face-attribute 'fixed-pitch nil
                      :family my-font-family)
  (set-face-attribute 'variable-pitch nil
                      :family my-variable-pitch-font-family)

  ;; --- One font size everywhere --------------------------------------------
  ;; Themes and packages declare their own face heights (org headings
  ;; at 1.3x, and so on). Force every named face to the relative 1.0:
  ;; no face may declare a size; all sizing flows from the default
  ;; face. (The startup screen's copyright uses an anonymous inline
  ;; face spec, not a named face — deliberately out of reach; see the
  ;; module prose.)
  (defun my-normalize-face-heights ()
    "Force every named face to the default face's height.
Sets each face's :height to the relative 1.0, overriding whatever
the theme or a package declared. Re-run after lazily loading a
package that introduces rogue face heights."
    (interactive)
    (dolist (face (face-list))
      (unless (eq face 'default)
        (ignore-errors
          (set-face-attribute face nil :height 1.0)))))

  (my-normalize-face-heights)
  ;; Faces defined later during init, and org's heading faces (org
  ;; loads lazily), get the same treatment.
  (add-hook 'after-init-hook #'my-normalize-face-heights)
  (with-eval-after-load 'org #'my-normalize-face-heights)

  ;; --- Zoom ------------------------------------------------------------------
  ;; Terminal-style: one knob, everything resizes at once.
  (defun my-apply-font-height ()
    "Apply `my-font-height' to the default face."
    (set-face-attribute 'default nil :height my-font-height))

  (defun my-zoom-in (&optional arg)
    "Increase the global font height by ARG steps (default 1).
Every face scales at once, like a terminal zoom."
    (interactive "p")
    (setq my-font-height
          (+ my-font-height
             (* my-font-height-step (or arg 1))))
    (my-apply-font-height))

  (defun my-zoom-out (&optional arg)
    "Decrease the global font height by ARG steps (default 1).
Never below one point."
    (interactive "p")
    (setq my-font-height
          (max my-font-height-step
               (- my-font-height
                  (* my-font-height-step (or arg 1)))))
    (my-apply-font-height))

  (defun my-zoom-reset ()
    "Reset the global font height to the default 9pt."
    (interactive)
    (setq my-font-height my-font-height-default)
    (my-apply-font-height))

  ;; --- Frame title ---------------------------------------------------------
  ;; Static: buffer identity is displayed elsewhere; the title never
  ;; needs to repeat it.
  (setopt frame-title-format '("GNU Emacs"))

  ;; --- Tooltips and dialogs ------------------------------------------------
  ;; Help text renders in the echo area; prompts go to the minibuffer.
  ;; Never a separate tooltip frame, never a GUI dialog box.
  (tooltip-mode -1)
  (setopt use-dialog-box nil)

  (provide 'my-emacs-frames)
#+end_src
